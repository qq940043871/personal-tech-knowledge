## 一、上下文窗口管理

```
上下文窗口挑战：
  - 模型有固定的最大token限制（如8K/32K/128K）
  - 对话越长，可用空间越少
  - 需要智能管理，保留关键信息

Token预算分配：
  ┌─────────────────────────────────────┐
  │  System Prompt     (~500 tokens)    │
  ├─────────────────────────────────────┤
  │  RAG Context       (~2000 tokens)   │
  ├─────────────────────────────────────┤
  │  Chat History      (~4000 tokens)   │
  ├─────────────────────────────────────┤
  │  Current Query     (~500 tokens)    │
  ├─────────────────────────────────────┤
  │  Reserved for Reply (~1000 tokens)  │
  └─────────────────────────────────────┘
```

## 二、滑动窗口策略

```python
from dataclasses import dataclass, field
from typing import List, Optional
from enum import Enum


class TruncationStrategy(Enum):
    KEEP_RECENT = "keep_recent"       # 保留最近N轮
    KEEP_IMPORTANT = "keep_important"  # 保留重要的
    SUMMARIZE_OLD = "summarize_old"    # 摘要旧对话


@dataclass
class ContextWindow:
    """上下文窗口管理器"""

    max_tokens: int = 8000
    system_tokens: int = 500
    reserved_reply: int = 1000
    strategy: TruncationStrategy = TruncationStrategy.KEEP_RECENT

    messages: List[dict] = field(default_factory=list)
    token_counts: List[int] = field(default_factory=list)

    @property
    def available_tokens(self) -> int:
        return self.max_tokens - self.system_tokens - self.reserved_reply

    @property
    def used_tokens(self) -> int:
        return sum(self.token_counts)

    def add_message(self, role: str, content: str, token_count: int,
                    importance: float = 0.5):
        """添加消息"""
        self.messages.append({
            "role": role,
            "content": content,
            "importance": importance,
        })
        self.token_counts.append(token_count)

        # 超出窗口时裁剪
        while self.used_tokens > self.available_tokens:
            self._truncate()

    def _truncate(self):
        """根据策略裁剪消息"""
        if self.strategy == TruncationStrategy.KEEP_RECENT:
            # 删除最旧的非system消息
            for i in range(len(self.messages)):
                if self.messages[i]["role"] != "system":
                    self.messages.pop(i)
                    self.token_counts.pop(i)
                    break

        elif self.strategy == TruncationStrategy.KEEP_IMPORTANT:
            # 删除最不重要的消息
            min_imp = float("inf")
            min_idx = -1
            for i, msg in enumerate(self.messages):
                if msg["role"] != "system" and msg["importance"] < min_imp:
                    min_imp = msg["importance"]
                    min_idx = i
            if min_idx >= 0:
                self.messages.pop(min_idx)
                self.token_counts.pop(min_idx)

    def get_context(self) -> List[dict]:
        """获取当前上下文"""
        return self.messages.copy()
```

## 三、上下文压缩

```python
class ContextCompressor:
    """上下文压缩器"""

    def __init__(self, llm):
        self.llm = llm

    async def summarize_history(self, messages: List[dict],
                                 max_summary_tokens: int = 300) -> str:
        """将历史对话压缩为摘要"""
        history_text = "\n".join(
            f"{m['role']}: {m['content']}" for m in messages
        )

        prompt = f"""请将以下对话历史压缩为简短摘要，保留关键信息：

{history_text}

摘要要求：
1. 保留用户的核心需求和关键决策
2. 保留重要的技术细节（API、变量名、文件路径）
3. 压缩到{max_summary_tokens} tokens以内
4. 使用简洁的要点格式

摘要："""

        summary = await self.llm.generate(prompt)
        return summary

    async def compress_context(self, context: ContextWindow) -> ContextWindow:
        """压缩上下文窗口"""
        if context.used_tokens <= context.available_tokens:
            return context

        # 将前半部分对话压缩为摘要
        mid = len(context.messages) // 2
        old_messages = context.messages[:mid]
        recent_messages = context.messages[mid:]

        summary = await self.summarize_history(old_messages)

        # 构建新的上下文
        new_ctx = ContextWindow(
            max_tokens=context.max_tokens,
            strategy=context.strategy,
        )
        new_ctx.add_message("system", f"历史摘要：{summary}", 300, importance=1.0)
        for msg in recent_messages:
            new_ctx.add_message(msg["role"], msg["content"],
                               context.token_counts[mid + recent_messages.index(msg)])

        return new_ctx
```
