## 一、混合检索架构

```
Hybrid Search架构：
  用户查询
      │
      ├─→ 向量检索（语义相似）
      │     文本 → Embedding → ANN搜索 → 结果集A
      │
      └─→ 关键词检索（精确匹配）
            文本 → 分词 → BM25/TF-IDF → 结果集B
      │
      └─→ 融合排序（RRF / 加权）
            结果集A + 结果集B → 最终排序 → Top-K

优势：
  - 向量检索：捕获语义相似（"快乐"≈"开心"）
  - 关键词检索：精确匹配专有名词、ID、代码
  - 互补：两者结合覆盖更多场景
```

## 二、RRF融合排序

```python
from typing import List, Dict, Tuple
from collections import defaultdict


class RRFFusion:
    """Reciprocal Rank Fusion（倒数排名融合）"""

    def __init__(self, k: int = 60):
        self.k = k  # 平滑常数，通常取60

    def fuse(self, *rankings: List[List[int]]) -> List[Tuple[int, float]]:
        """
        融合多个排序结果

        Args:
            rankings: 多个排序结果，每个是 [doc_id, ...] 的列表

        Returns:
            融合后的排序结果 [(doc_id, rrf_score), ...]
        """
        scores: Dict[int, float] = defaultdict(float)

        for ranking in rankings:
            for rank, doc_id in enumerate(ranking, start=1):
                scores[doc_id] += 1.0 / (self.k + rank)

        # 按分数降序排序
        sorted_results = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return sorted_results


class HybridSearcher:
    """混合检索器"""

    def __init__(self, vector_store, bm25_index):
        self.vector_store = vector_store
        self.bm25_index = bm25_index
        self.fusion = RRFFusion(k=60)

    async def search(self, query: str, query_embedding: List[float],
                     top_k: int = 10, alpha: float = 0.5) -> List[Dict]:
        """
        混合检索

        Args:
            query: 原始查询文本
            query_embedding: 查询的向量表示
            top_k: 返回数量
            alpha: 向量权重（1-alpha为关键词权重）
        """
        # 1. 向量检索
        vector_results = self.vector_store.search(
            query_embedding, k=top_k * 2
        )
        vector_ids = [r[0] for r in vector_results]

        # 2. 关键词检索
        keyword_results = self.bm25_index.search(query, k=top_k * 2)
        keyword_ids = [r[0] for r in keyword_results]

        # 3. RRF融合
        fused = self.fusion.fuse(vector_ids, keyword_ids)

        # 4. 返回Top-K
        return [{"doc_id": doc_id, "score": score}
                for doc_id, score in fused[:top_k]]
```

## 三、BM25关键词检索

```python
import math
from collections import Counter
from typing import List, Dict


class BM25Index:
    """BM25关键词索引"""

    def __init__(self, k1: float = 1.5, b: float = 0.75):
        self.k1 = k1
        self.b = b
        self.doc_lengths: Dict[int, int] = {}
        self.avg_doc_length: float = 0
        self.doc_freqs: Dict[str, int] = {}  # 词 → 出现该词的文档数
        self.doc_term_freqs: Dict[int, Counter] = {}
        self.total_docs: int = 0

    def add(self, doc_id: int, text: str):
        """添加文档"""
        terms = list(text)  # 简化：按字符分词
        self.doc_lengths[doc_id] = len(terms)
        self.doc_term_freqs[doc_id] = Counter(terms)

        for term in set(terms):
            self.doc_freqs[term] = self.doc_freqs.get(term, 0) + 1

        self.total_docs += 1
        self.avg_doc_length = sum(self.doc_lengths.values()) / self.total_docs

    def search(self, query: str, k: int = 10) -> List[Tuple[int, float]]:
        """BM25检索"""
        query_terms = list(query)
        scores: Dict[int, float] = defaultdict(float)

        for term in query_terms:
            if term not in self.doc_freqs:
                continue

            df = self.doc_freqs[term]
            idf = math.log((self.total_docs - df + 0.5) / (df + 0.5) + 1)

            for doc_id, term_freqs in self.doc_term_freqs.items():
                tf = term_freqs.get(term, 0)
                if tf == 0:
                    continue
                dl = self.doc_lengths[doc_id]
                numerator = tf * (self.k1 + 1)
                denominator = tf + self.k1 * (1 - self.b + self.b * dl / self.avg_doc_length)
                scores[doc_id] += idf * numerator / denominator

        sorted_results = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return sorted_results[:k]
```
