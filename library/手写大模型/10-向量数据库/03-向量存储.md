## 一、向量持久化方案

```
向量存储架构：
  内存层：
    - 热数据，支持实时查询
    - HNSW/IVF索引常驻内存
    - 占用：维度 × 4字节 × 向量数

  磁盘层：
    - 冷数据持久化
    - WAL（Write-Ahead Log）保证一致性
    - LSM-Tree或B+Tree管理

  存储引擎选型：
    - RocksDB：嵌入式KV，适合单机
    - Milvus：分布式向量数据库
    - Qdrant：Rust实现，性能优秀
    - Chroma：轻量级，适合开发测试
```

## 二、向量存储引擎

```python
import struct
import numpy as np
from pathlib import Path
from typing import List, Tuple, Optional, Dict


class VectorStore:
    """向量存储引擎"""

    def __init__(self, dimension: int, store_path: str):
        self.dimension = dimension
        self.store_path = Path(store_path)
        self.store_path.mkdir(parents=True, exist_ok=True)
        self.vectors_file = self.store_path / "vectors.bin"
        self.meta_file = self.store_path / "meta.bin"
        self.index: Dict[int, int] = {}  # id → offset

    def add(self, vector: np.ndarray, doc_id: int, metadata: Dict = None):
        """添加向量到存储"""
        # 追加到向量文件
        offset = self.vectors_file.stat().st_size if self.vectors_file.exists() else 0
        with open(self.vectors_file, "ab") as f:
            f.write(vector.astype(np.float32).tobytes())

        # 记录元数据
        with open(self.meta_file, "ab") as f:
            meta_bytes = str(metadata or {}).encode("utf-8")
            f.write(struct.pack("I", len(meta_bytes)))
            f.write(meta_bytes)

        self.index[doc_id] = offset

    def get(self, doc_id: int) -> Optional[np.ndarray]:
        """根据ID获取向量"""
        if doc_id not in self.index:
            return None
        offset = self.index[doc_id]
        with open(self.vectors_file, "rb") as f:
            f.seek(offset)
            data = f.read(self.dimension * 4)
            return np.frombuffer(data, dtype=np.float32)

    def batch_add(self, vectors: np.ndarray, doc_ids: List[int],
                  metadatas: List[Dict] = None):
        """批量添加向量"""
        with open(self.vectors_file, "ab") as f_meta, \
             open(self.meta_file, "ab") as f_vec:
            for i, (vec, did) in enumerate(zip(vectors, doc_ids)):
                offset = self.vectors_file.stat().st_size
                f_vec.write(vec.astype(np.float32).tobytes())
                meta_bytes = str(metadatas[i] if metadatas else {}).encode("utf-8")
                f_meta.write(struct.pack("I", len(meta_bytes)))
                f_meta.write(meta_bytes)
                self.index[did] = offset

    def load_all(self) -> Tuple[np.ndarray, List[int]]:
        """加载所有向量到内存"""
        if not self.vectors_file.exists():
            return np.array([]), []
        data = np.fromfile(str(self.vectors_file), dtype=np.float32)
        vectors = data.reshape(-1, self.dimension)
        doc_ids = list(self.index.keys())
        return vectors, doc_ids
```

## 三、分片与副本

```python
class ShardedVectorStore:
    """分片向量存储"""

    def __init__(self, dimension: int, num_shards: int = 4):
        self.dimension = dimension
        self.num_shards = num_shards
        self.shards: List[VectorStore] = []

    def _get_shard(self, doc_id: int) -> int:
        """根据doc_id确定所属分片"""
        return doc_id % self.num_shards

    def add(self, vector: np.ndarray, doc_id: int, metadata: Dict = None):
        shard_idx = self._get_shard(doc_id)
        self.shards[shard_idx].add(vector, doc_id, metadata)

    def search(self, query: np.ndarray, k: int = 10) -> List[Tuple[int, float]]:
        """跨分片搜索"""
        all_results = []
        for shard in self.shards:
            vectors, doc_ids = shard.load_all()
            if len(vectors) == 0:
                continue
            # 计算距离
            distances = np.linalg.norm(vectors - query, axis=1)
            for i, dist in enumerate(distances):
                all_results.append((doc_ids[i], float(dist)))

        # 合并排序取top-k
        all_results.sort(key=lambda x: x[1])
        return all_results[:k]
```
