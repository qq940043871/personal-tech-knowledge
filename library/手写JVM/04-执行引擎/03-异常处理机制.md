## 一、异常处理概述

JVM的异常处理基于异常表（Exception Table），而非传统的栈展开。每个方法的Code属性中包含异常表，记录了try-catch的范围和处理位置。

```
异常表结构：
┌──────────┬──────────┬────────────┬───────────┐
│ startPc  │  endPc   │ handlerPc  │ catchType │
├──────────┼──────────┼────────────┼───────────┤
│    0     │    4     │     8      │  Exception │
└──────────┴──────────┴────────────┴───────────┘
```

## 二、异常表结构

```java
public class ExceptionTable {

    public static class Entry {
        public int startPc;      // try块起始PC
        public int endPc;        // try块结束PC
        public int handlerPc;    // catch处理器PC
        public int catchType;    // 捕获的异常类型（常量池索引，0表示finally）

        public Entry(int startPc, int endPc, int handlerPc, int catchType) {
            this.startPc = startPc;
            this.endPc = endPc;
            this.handlerPc = handlerPc;
            this.catchType = catchType;
        }
    }

    private Entry[] entries;
    private RuntimeConstantPool constantPool;

    // 查找匹配的异常处理器
    public Entry findHandler(int currentPc, Clazz exceptionClass) {
        for (Entry entry : entries) {
            // 检查PC是否在try块范围内
            if (currentPc >= entry.startPc && currentPc < entry.endPc) {
                // catchType为0表示catch-all（finally）
                if (entry.catchType == 0) {
                    return entry;
                }
                // 检查异常类型是否匹配
                Clazz catchClass = resolveCatchType(entry.catchType);
                if (isAssignable(exceptionClass, catchClass)) {
                    return entry;
                }
            }
        }
        return null; // 没有找到处理器
    }

    private Clazz resolveCatchType(int index) {
        if (index == 0) return null; // finally
        ConstantClassInfo info = (ConstantClassInfo) constantPool.get(index);
        return classLoader.loadClass(info.getClassName());
    }
}
```

## 三、try-catch编译实现

```java
public class TryCatchCompiler {

    // 源码：
    // try {
    //     int a = 1 / b;
    // } catch (ArithmeticException e) {
    //     System.out.println("error");
    // } finally {
    //     System.out.println("done");
    // }

    // 编译后的字节码：
    // 0: iconst_1
    // 1: iload_1
    // 2: idiv
    // 3: istore_2
    // 4: goto 20
    // 7: astore_2         // catch ArithmeticException
    // 8: getstatic System.out
    // 11: ldc "error"
    // 13: invokevirtual println
    // 16: goto 20
    // 19: astore_3        // finally (any exception)
    // 20: getstatic System.out
    // 23: ldc "done"
    // 25: invokevirtual println
    // 28: return

    public ExceptionTable compileTryCatch(TryCatchNode node) {
        ExceptionTable table = new ExceptionTable();

        // 编译try块
        int startPc = currentPC();
        compile(node.tryBlock);
        int endPc = currentPC();

        // 编译跳转到finally
        emitGoto(node.finallyBlock.startPc);

        // 编译catch块
        for (CatchClause catchClause : node.catchClauses) {
            int handlerPc = currentPC();
            compile(catchClause.body);
            // 注册异常表项
            table.addEntry(new ExceptionTable.Entry(
                startPc, endPc, handlerPc, catchClause.exceptionType
            ));
            emitGoto(node.finallyBlock.startPc);
        }

        // 编译finally块（复制到多个位置）
        int finallyPc = currentPC();
        compile(node.finallyBlock);
        // finally块也需要注册异常表
        table.addEntry(new ExceptionTable.Entry(
            startPc, endPc, finallyPc, 0 // catchType=0 表示catch-all
        ));

        return table;
    }
}
```

## 四、异常抛出实现

```java
public class ExceptionThrower {

    // athrow指令实现
    public void executeAThrow(ThreadContext context) {
        // 1. 从操作数栈弹出异常对象
        Object exception = context.getCurrentFrame().popRef();
        if (exception == null) {
            throw new NullPointerException();
        }

        // 2. 在当前方法的异常表中查找处理器
        throwException(context, exception);
    }

    private void throwException(ThreadContext context, Object exception) {
        while (true) {
            StackFrame frame = context.getCurrentFrame();
            int currentPC = context.getPC().get();
            ExceptionTable table = frame.getMethod().getExceptionTable();

            // 查找匹配的处理器
            ExceptionTable.Entry handler = table.findHandler(
                currentPC, exception.getClass()
            );

            if (handler != null) {
                // 找到处理器：跳转到handlerPc
                context.getPC().set(handler.handlerPc);

                // 清空操作数栈
                frame.clearOperandStack();

                // 压入异常对象
                frame.pushRef(exception);
                return;
            }

            // 没找到：方法异常退出，栈展开
            StackFrame caller = frame.getCaller();
            if (caller == null) {
                // 所有栈帧都已弹出，线程终止
                handleUncaughtException(context, exception);
                return;
            }

            // 弹出当前栈帧，继续在调用者中查找
            context.setCurrentFrame(caller);
            context.getPC().set(frame.getReturnAddress());
        }
    }

    // 未捕获异常处理
    private void handleUncaughtException(ThreadContext context, Object exception) {
        Thread thread = context.getThread();
        UncaughtExceptionHandler handler = thread.getUncaughtExceptionHandler();
        if (handler != null) {
            handler.uncaughtException(thread, (Throwable) exception);
        } else {
            System.err.println("Exception in thread \"" + thread.getName() + "\"");
            ((Throwable) exception).printStackTrace(System.err);
        }
    }
}
```

## 五、栈展开过程

```java
public class StackUnwinding {

    // 栈展开：当异常未在当前方法捕获时，逐层弹出栈帧
    public void unwind(ThreadContext context, Object exception) {
        while (true) {
            StackFrame current = context.getCurrentFrame();
            if (current == null) {
                // 栈已空，未捕获异常
                throw new UncaughtException(exception);
            }

            // 执行finally块（如果有）
            executeFinallyIfAny(current);

            // 检查是否有匹配的catch
            ExceptionTable.Entry handler = findHandler(current, exception);
            if (handler != null) {
                // 找到处理器
                context.getPC().set(handler.handlerPc);
                current.clearOperandStack();
                current.pushRef(exception);
                return;
            }

            // 弹出当前帧，继续展开
            StackFrame caller = current.getCaller();
            context.setCurrentFrame(caller);
            if (caller != null) {
                context.getPC().set(current.getReturnAddress());
            }
        }
    }

    // finally块的特殊处理
    private void executeFinallyIfAny(StackFrame frame) {
        ExceptionTable table = frame.getMethod().getExceptionTable();
        for (ExceptionTable.Entry entry : table) {
            if (entry.catchType == 0) { // finally
                int savedPC = frame.getPC().get();
                frame.getPC().set(entry.handlerPc);
                // 执行finally代码
                executeCode(frame, entry.handlerPc);
                frame.getPC().set(savedPC);
            }
        }
    }
}
```

## 六、常见异常类型

```java
public class CommonExceptions {

    // NullPointerException
    public void npeExample() {
        Object obj = null;
        obj.toString(); // 抛出NPE
    }

    // ArrayIndexOutOfBoundsException
    public void aioobExample() {
        int[] arr = new int[5];
        arr[10] = 1; // 抛出AIOOBE
    }

    // ClassCastException
    public void cceExample() {
        Object obj = "hello";
        Integer i = (Integer) obj; // 抛出CCE
    }

    // StackOverflowError
    public void soeExample() {
        soeExample(); // 递归导致栈溢出
    }

    // OutOfMemoryError
    public void oomExample() {
        List<byte[]> list = new ArrayList<>();
        while (true) {
            list.add(new byte[1024 * 1024]); // 堆溢出
        }
    }
}
```
