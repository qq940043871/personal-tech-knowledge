## 一、方法调用指令概述

JVM提供了5条方法调用指令，对应不同的调用场景：

| 指令 | 说明 | 场景 |
|------|------|------|
| invokestatic | 调用静态方法 | 类方法调用 |
| invokespecial | 调用构造器/私有方法/父类方法 | <init>、super调用 |
| invokevirtual | 调用虚方法 | 普通实例方法 |
| invokeinterface | 调用接口方法 | 接口方法调用 |
| invokedynamic | 动态调用 | Lambda、方法引用 |

## 二、invokestatic实现

```java
public class InvokeStatic {

    // invokestatic：调用静态方法，无需多态查找
    public void execute(StackFrame frame, int constantPoolIndex) {
        // 1. 从常量池解析方法引用
        Method method = resolveMethod(constantPoolIndex);

        // 2. 验证是静态方法
        if (!method.isStatic()) {
            throw new IncompatibleClassChangeError();
        }

        // 3. 确保方法所属类已初始化
        if (!method.getClazz().isInitialized()) {
            initializeClass(method.getClazz());
        }

        // 4. 创建新栈帧
        StackFrame newFrame = new StackFrame(method, frame);

        // 5. 传递参数（无this）
        passArguments(frame, newFrame, method);

        // 6. 压入栈帧
        pushFrame(newFrame);
    }
}
```

## 三、invokevirtual实现

```java
public class InvokeVirtual {

    // invokevirtual：虚方法调用，支持多态
    public void execute(StackFrame frame, int constantPoolIndex) {
        // 1. 解析方法符号引用
        MethodRef ref = resolveMethodRef(constantPoolIndex);
        String methodName = ref.getName();
        String descriptor = ref.getDescriptor();

        // 2. 从操作数栈获取this引用
        Object receiver = frame.peekRef(ref.getParamSlotCount());
        if (receiver == null) {
            throw new NullPointerException();
        }

        // 3. 多态分派：根据实际类型查找方法
        Method actualMethod = lookupVirtualMethod(
            receiver.getClass(), methodName, descriptor
        );

        // 4. 创建栈帧并调用
        StackFrame newFrame = new StackFrame(actualMethod, frame);
        passArguments(frame, newFrame, actualMethod);
        pushFrame(newFrame);
    }

    // 虚方法表查找
    private Method lookupVirtualMethod(Class<?> clazz, String name, String desc) {
        VirtualMethodTable vmt = getMethodTable(clazz);
        return vmt.lookup(name, desc);
    }
}
```

## 四、虚方法表（vtable）

```java
public class VirtualMethodTable {

    // 虚方法表：加速虚方法分派
    private Method[] table;
    private int size;

    public VirtualMethodTable(Clazz clazz) {
        // 1. 复制父类的虚方法表
        VirtualMethodTable parent = clazz.getSuperClass().getVtable();
        this.size = parent.size;
        this.table = Arrays.copyOf(parent.table, parent.size);

        // 2. 覆盖或新增方法
        for (Method method : clazz.getMethods()) {
            if (!method.isStatic() && !method.isPrivate() && !method.isConstructor()) {
                int index = findSlot(method.getName(), method.getDescriptor());
                if (index >= 0) {
                    // 覆盖父类方法
                    table[index] = method;
                } else {
                    // 新增方法
                    table[size++] = method;
                }
            }
        }
    }

    // 查找虚方法
    public Method lookup(String name, String descriptor) {
        for (int i = 0; i < size; i++) {
            if (table[i].getName().equals(name)
                    && table[i].getDescriptor().equals(descriptor)) {
                return table[i];
            }
        }
        return null;
    }

    private int findSlot(String name, String desc) {
        for (int i = 0; i < size; i++) {
            if (table[i].getName().equals(name)
                    && table[i].getDescriptor().equals(desc)) {
                return i;
            }
        }
        return -1;
    }
}
```

## 五、invokeinterface实现

```java
public class InvokeInterface {

    // invokeinterface：接口方法调用
    public void execute(StackFrame frame, int constantPoolIndex) {
        // 1. 解析接口方法引用
        MethodRef ref = resolveInterfaceMethodRef(constantPoolIndex);

        // 2. 获取接收者对象
        Object receiver = frame.peekRef(ref.getParamSlotCount());
        if (receiver == null) {
            throw new NullPointerException();
        }

        // 3. 在实际类中查找方法
        Method actualMethod = lookupInterfaceMethod(
            receiver.getClass(), ref.getName(), ref.getDescriptor()
        );

        // 4. 调用
        StackFrame newFrame = new StackFrame(actualMethod, frame);
        passArguments(frame, newFrame, actualMethod);
        pushFrame(newFrame);
    }

    // 接口方法表（itable）查找
    private Method lookupInterfaceMethod(Class<?> clazz, String name, String desc) {
        InterfaceMethodTable itable = getInterfaceMethodTable(clazz);
        return itable.lookup(name, desc);
    }
}
```

## 六、方法返回实现

```java
public class MethodReturn {

    // return指令（void）
    public void executeReturn(ThreadContext context) {
        StackFrame current = context.getCurrentFrame();
        context.setCurrentFrame(current.getCaller());
        context.getPC().set(current.getReturnAddress());
    }

    // ireturn指令（返回int）
    public void executeIReturn(ThreadContext context) {
        StackFrame current = context.getCurrentFrame();
        int value = current.popInt();

        StackFrame caller = current.getCaller();
        context.setCurrentFrame(caller);
        caller.pushInt(value);
        context.getPC().set(current.getReturnAddress());
    }

    // areturn指令（返回引用）
    public void executeAReturn(ThreadContext context) {
        StackFrame current = context.getCurrentFrame();
        Object ref = current.popRef();

        StackFrame caller = current.getCaller();
        context.setCurrentFrame(caller);
        caller.pushRef(ref);
        context.getPC().set(current.getReturnAddress());
    }

    // lreturn指令（返回long）
    public void executeLReturn(ThreadContext context) {
        StackFrame current = context.getCurrentFrame();
        long value = current.popLong();

        StackFrame caller = current.getCaller();
        context.setCurrentFrame(caller);
        caller.pushLong(value);
        context.getPC().set(current.getReturnAddress());
    }
}
```

## 七、方法分派总结

```java
public class MethodDispatch {

    // 静态分派：编译期确定，基于静态类型
    // 重载(Overload)是静态分派
    public void staticDispatch() {
        Human man = new Man();
        Human woman = new Woman();
        // 编译期根据静态类型Human选择方法
        sayHello(man);    // 输出: hello human
        sayHello(woman);  // 输出: hello human
    }

    void sayHello(Human h) { System.out.println("hello human"); }
    void sayHello(Man m) { System.out.println("hello man"); }
    void sayHello(Woman w) { System.out.println("hello woman"); }

    // 动态分派：运行期确定，基于实际类型
    // 重写(Override)是动态分派
    public void dynamicDispatch() {
        Human man = new Man();
        Human woman = new Woman();
        // 运行期根据实际类型选择方法
        man.say();    // 输出: man say
        woman.say();  // 输出: woman say
    }
}
```

## 八、总结

方法调用是JVM执行引擎的核心功能，理解invoke*指令和分派机制是实现JVM的关键。

**学习要点**：
1. 掌握5条invoke指令的使用场景
2. 理解虚方法表的构建和查找过程
3. 掌握静态分派与动态分派的区别
4. 理解方法返回时栈帧的切换
