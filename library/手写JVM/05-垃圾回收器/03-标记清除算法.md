## 一、标记清除算法概述

标记清除（Mark-Sweep）是最基础的垃圾回收算法，分为两个阶段：标记所有存活对象，然后清除未标记的对象。

```
标记阶段：
┌───┬───┬───┬───┬───┬───┬───┬───┐
│ A │ B │ C │ D │ E │ F │ G │ H │
│ ✓ │   │ ✓ │   │ ✓ │   │ ✓ │   │
└───┴───┴───┴───┴───┴───┴───┴───┘

清除阶段：
┌───┬───┬───┬───┬───┬───┬───┬───┐
│ A │   │ C │   │ E │   │ G │   │
│   │ 空│   │ 空│   │ 空│   │ 空│
└───┴───┴───┴───┴───┴───┴───┴───┘

结果：产生内存碎片
```

## 二、标记清除实现

```java
public class MarkSweepCollector {

    private byte[] heap;
    private int heapSize;
    private Map<Integer, ObjectHeader> objectMap;

    // 对象头
    public static class ObjectHeader {
        int address;
        int size;
        boolean marked;
        ObjectHeader next; // 空闲链表指针
    }

    // 标记阶段
    public void mark(Set<Object> roots) {
        Deque<Object> worklist = new ArrayDeque<>(roots);

        while (!worklist.isEmpty()) {
            Object obj = worklist.pop();
            ObjectHeader header = getHeader(obj);
            if (header == null || header.marked) continue;

            // 标记为存活
            header.marked = true;

            // 遍历引用字段
            List<Object> refs = getReferences(obj);
            for (Object ref : refs) {
                if (ref != null) {
                    worklist.push(ref);
                }
            }
        }
    }

    // 清除阶段
    public void sweep() {
        ObjectHeader current = head;
        while (current != null) {
            if (!current.marked) {
                // 未标记的对象，释放内存
                freeObject(current);
            } else {
                // 存活对象，取消标记（为下次GC准备）
                current.marked = false;
            }
            current = current.next;
        }
    }

    // 完整的GC流程
    public void collect() {
        // 1. 收集GC Roots
        Set<Object> roots = gcRoots.collectRoots();

        // 2. 标记存活对象
        mark(roots);

        // 3. 清除不可达对象
        sweep();

        // 4. 整理空闲链表
        coalesce();
    }

    private void freeObject(ObjectHeader header) {
        // 将释放的空间加入空闲链表
        header.next = freeList;
        freeList = header;
        totalFree += header.size;
    }
}
```

## 三、空闲链表管理

```java
public class FreeListManager {

    // 空闲链表节点
    public static class FreeBlock {
        int address;
        int size;
        FreeBlock next;

        public FreeBlock(int address, int size) {
            this.address = address;
            this.size = size;
        }
    }

    private FreeBlock head;

    // 首次适配分配
    public int allocateFirstFit(int size) {
        FreeBlock prev = null;
        FreeBlock current = head;

        while (current != null) {
            if (current.size >= size) {
                // 找到合适大小的块
                int addr = current.address;
                if (current.size > size) {
                    // 分割块
                    current.address += size;
                    current.size -= size;
                } else {
                    // 完全使用
                    if (prev != null) {
                        prev.next = current.next;
                    } else {
                        head = current.next;
                    }
                }
                return addr;
            }
            prev = current;
            current = current.next;
        }
        return -1; // 空间不足
    }

    // 最佳适配分配
    public int allocateBestFit(int size) {
        FreeBlock best = null;
        FreeBlock bestPrev = null;
        FreeBlock prev = null;
        FreeBlock current = head;

        while (current != null) {
            if (current.size >= size) {
                if (best == null || current.size < best.size) {
                    best = current;
                    bestPrev = prev;
                }
            }
            prev = current;
            current = current.next;
        }

        if (best != null) {
            int addr = best.address;
            if (best.size > size) {
                best.address += size;
                best.size -= size;
            } else {
                if (bestPrev != null) {
                    bestPrev.next = best.next;
                } else {
                    head = best.next;
                }
            }
            return addr;
        }
        return -1;
    }

    // 合并相邻空闲块
    public void coalesce() {
        FreeBlock current = head;
        while (current != null && current.next != null) {
            if (current.address + current.size == current.next.address) {
                // 合并相邻块
                current.size += current.next.size;
                current.next = current.next.next;
            } else {
                current = current.next;
            }
        }
    }
}
```

## 四、内存碎片问题

```java
public class FragmentationAnalyzer {

    // 分析内存碎片率
    public double calculateFragmentation(FreeBlock freeList) {
        int totalFree = 0;
        int largestFree = 0;
        int freeBlocks = 0;

        FreeBlock current = freeList;
        while (current != null) {
            totalFree += current.size;
            largestFree = Math.max(largestFree, current.size);
            freeBlocks++;
            current = current.next;
        }

        if (totalFree == 0) return 0;

        // 碎片率 = 1 - (最大空闲块 / 总空闲空间)
        double fragmentation = 1.0 - (double) largestFree / totalFree;
        return fragmentation;
    }

    // 碎片导致分配失败示例
    public void fragmentExample() {
        // 假设堆总大小100字节，空闲40字节但分散在4个10字节的块中
        // 申请30字节会失败，尽管总空闲足够
        FreeBlock f1 = new FreeBlock(10, 10);
        FreeBlock f2 = new FreeBlock(30, 10);
        FreeBlock f3 = new FreeBlock(50, 10);
        FreeBlock f4 = new FreeBlock(70, 10);
        f1.next = f2; f2.next = f3; f3.next = f4;

        int addr = allocateFirstFit(30); // 返回-1，分配失败！
    }

    // 解决方案
    // 1. 标记-整理算法：移动存活对象消除碎片
    // 2. 复制算法：将存活对象复制到新区域
    // 3. 分代收集：不同代使用不同算法
}
```

## 五、标记清除的STW问题

```java
public class STWManager {

    // Stop-The-World：GC时暂停所有应用线程
    private volatile boolean gcInProgress = false;

    // 完整的STW GC流程
    public void collectWithSTW() {
        // 1. 暂停所有应用线程
        safepoint.requestSafepoint();
        gcInProgress = true;

        try {
            // 2. 标记阶段（STW）
            Set<Object> roots = gcRoots.collectAllRoots();
            mark(roots);

            // 3. 清除阶段（STW）
            sweep();

        } finally {
            // 4. 恢复所有应用线程
            gcInProgress = false;
            safepoint.releaseSafepoint();
        }
    }

    // 减少STW时间的优化
    // 1. 并发标记：标记阶段与应用线程并发执行
    // 2. 增量标记：分多次小批量标记
    // 3. 分代收集：只对部分代进行GC
}
```

## 六、标记清除算法的优缺点

```java
public class MarkSweepAnalysis {

    // 优点：
    // 1. 实现简单
    // 2. 不需要移动对象
    // 3. 不需要额外的内存空间（与复制算法相比）

    // 缺点：
    // 1. 效率不高：标记和清除都需要遍历
    // 2. 产生内存碎片

    // 4. STW时间长

    // 适用场景：
    // - 老年代：对象存活率高，复制算法效率低
    // - 配合标记-整理使用
}
```

## 七、总结

标记清除是垃圾回收的基础算法，虽然有内存碎片问题，但为后续的复制算法、标记整理算法奠定了基础。

**学习要点**：
1. 掌握标记清除的两阶段执行过程
2. 理解空闲链表的管理方式
3. 掌握内存碎片的产生原因和解决方案
4. 理解STW对应用性能的影响
