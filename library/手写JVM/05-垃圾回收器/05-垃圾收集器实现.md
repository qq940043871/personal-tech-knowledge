## 一、垃圾收集器概述

JVM有多种垃圾收集器，针对不同场景优化。主要的收集器包括CMS、G1和ZGC。

```
收集器演进：
Serial → Parallel → CMS → G1 → ZGC
                    ↓
               停顿时间优先
```

## 二、CMS收集器

```java
public class CMSCollector {

    // CMS (Concurrent Mark-Sweep) 收集器
    // 目标：最短停顿时间

    // 四个阶段
    public void collect() {
        // 阶段1：初始标记 (STW)
        initialMark();

        // 阶段2：并发标记
        concurrentMark();

        // 阶段3：重新标记 (STW)
        remark();

        // 阶段4：并发清除
        concurrentSweep();
    }

    // 初始标记：标记GC Roots直接引用的对象（STW）
    private void initialMark() {
        safepoint.request();
        // 只标记GC Roots直接引用，速度很快
        Set<Object> roots = gcRoots.collectRoots();
        for (Object root : roots) {
            markDirect(root);
        }
        safepoint.release();
    }

    // 并发标记：与应用线程并发执行
    private void concurrentMark() {
        // 使用写屏障记录引用变更
        // 遍历对象图标记所有存活对象
        Deque<Object> worklist = new ArrayDeque<>(initialMarkSet);
        while (!worklist.isEmpty()) {
            Object obj = worklist.pop();
            for (Object ref : getReferences(obj)) {
                if (ref != null && !isMarked(ref)) {
                    mark(ref);
                    worklist.push(ref);
                }
            }
        }
    }

    // 重新标记：修正并发标记期间的引用变更（STW）
    private void remark() {
        safepoint.request();
        // 处理写屏障记录的变更
        Set<Object> dirtyCards = cardTable.getDirtyCards();
        for (Object card : dirtyCards) {
            rescanObject(card);
        }
        safepoint.release();
    }

    // 并发清除：清除未标记的对象
    private void concurrentSweep() {
        // 遍历整个堆，回收未标记的对象
        for (int addr = heapStart; addr < heapEnd; addr += getObjectSize(addr)) {
            if (!isMarked(addr)) {
                freeObject(addr);
            } else {
                clearMark(addr);
            }
        }
    }
}
```

## 三、G1收集器

```java
public class G1Collector {

    // G1 (Garbage-First) 收集器
    // 将堆划分为多个Region，优先回收垃圾最多的Region

    private static final int REGION_SIZE = 1024 * 1024; // 1MB
    private Region[] regions;

    public static class Region {
        int start;
        int size;
        RegionType type;  // Eden, Survivor, Old, Humongous
        long garbageBytes; // 垃圾字节数
        boolean selected;  // 是否被选入回收集
    }

    enum RegionType { EDEN, SURVIVOR, OLD, HUMONGOUS, FREE }

    // G1收集流程
    public void collect() {
        // 1. 初始标记 (STW)
        initialMark();

        // 2. 并发标记
        concurrentMark();

        // 3. 最终标记 (STW)
        finalMark();

        // 4. 筛选回收 (STW)
        cleanup();
    }

    // 筛选回收：选择垃圾最多的Region
    private void cleanup() {
        // 计算每个Region的垃圾比例
        List<Region> candidates = new ArrayList<>();
        for (Region region : regions) {
            if (region.type == RegionType.OLD) {
                region.garbageBytes = calculateGarbage(region);
                candidates.add(region);
            }
        }

        // 按垃圾比例排序，优先回收垃圾最多的
        candidates.sort((a, b) ->
            Long.compare(b.garbageBytes, a.garbageBytes));

        // 选择回收集，在停顿时间目标内尽可能多回收
        long maxPause = maxPauseTimeMillis;
        long estimatedPause = 0;
        Set<Region> collectionSet = new HashSet<>();

        for (Region region : candidates) {
            long regionPause = estimatePauseTime(region);
            if (estimatedPause + regionPause <= maxPause) {
                collectionSet.add(region);
                estimatedPause += regionPause;
            } else {
                break;
            }
        }

        // 执行回收
        evacuateCollectionSet(collectionSet);
    }

    // RSet (Remembered Set)：记录跨Region引用
    private Map<Region, Set<Integer>> rememberedSets;

    public void updateRSet(Region from, Region to, int refAddr) {
        rememberedSets.computeIfAbsent(to, k -> new HashSet<>())
                      .add(refAddr);
    }
}
```

## 四、ZGC收集器

```java
public class ZGCCollector {

    // ZGC：超低延迟收集器（停顿时间 < 10ms）
    // 使用着色指针和读屏障技术

    // 着色指针：利用64位指针的高位存储GC元数据
    // 位 0-41: 对象地址
    // 位 42:   Finalizable标记
    // 位 43:   Remapped标记
    // 位 44:   Marked0标记
    // 位 45:   Marked1标记

    public static class ZPointer {
        static final long ADDRESS_MASK   = 0x0000FFFFFFFFFFFFL;
        static final long FINALIZABLE    = 1L << 42;
        static final long REMAPPED       = 1L << 43;
        static final long MARKED0        = 1L << 44;
        static final long MARKED1        = 1L << 45;

        public static long extractAddress(long pointer) {
            return pointer & ADDRESS_MASK;
        }

        public static boolean isMarked(long pointer) {
            return (pointer & (MARKED0 | MARKED1)) != 0;
        }

        public static boolean isRemapped(long pointer) {
            return (pointer & REMAPPED) != 0;
        }
    }

    // ZGC收集流程（几乎完全并发）
    public void collect() {
        // 1. 初始标记 (STW，仅标记GC Roots)
        pauseAndMarkRoots();

        // 2. 并发标记
        concurrentMark();

        // 3. 并发预备重分配
        concurrentPrepareForRelocate();

        // 4. 初始转移 (STW，仅转移GC Roots)
        pauseAndRelocateRoots();

        // 5. 并发重分配
        concurrentRelocate();

        // 6. 并发重映射
        concurrentRemap();
    }

    // 读屏障：应用线程读取引用时检查着色指针
    public Object loadBarrier(Object refField) {
        long pointer = toPointer(refField);
        if (!ZPointer.isRemapped(pointer)) {
            // 需要重映射
            return relocateAndRemap(pointer);
        }
        return refField;
    }

    // 并发重分配：将对象转移到新的Region
    private void concurrentRelocate() {
        for (ZPage page : relocationSet) {
            for (long addr = page.start(); addr < page.end(); addr += objectSize(addr)) {
                if (isLive(addr)) {
                    long newAddr = allocateInTargetPage(getSize(addr));
                    copyObject(addr, newAddr);
                    updateForwarding(addr, newAddr);
                }
            }
        }
    }
}
```

## 五、收集器对比

```java
public class CollectorComparison {

    // CMS vs G1 vs ZGC 对比
    /*
    ┌──────────┬──────────┬──────────┬──────────┐
    │   特性    │   CMS    │    G1    │   ZGC    │
    ├──────────┼──────────┼──────────┼──────────┤
    │ 目标     │ 低停顿   │ 均衡     │ 超低停顿 │
    │ 停顿时间 │ 100ms+   │ 50ms     │ <10ms    │
    │ 堆大小   │ <4GB     │ <64GB    │ <16TB    │
    │ 碎片     │ 有碎片   │ 无碎片   │ 无碎片   │
    │ 并发度   │ 部分并发 │ 部分并发 │ 几乎全并发│
    │ 算法     │ Mark-Sweep│ 分Region │ 着色指针 │
    └──────────┴──────────┴──────────┴──────────┘
    */

    // 收集器选择建议
    public static String recommendCollector(long heapSize, long maxPause) {
        if (heapSize > 16L * 1024 * 1024 * 1024) {
            return "ZGC"; // 超大堆
        }
        if (maxPause < 10) {
            return "ZGC"; // 超低延迟
        }
        if (maxPause < 50) {
            return "G1"; // 低延迟
        }
        if (heapSize < 4L * 1024 * 1024 * 1024) {
            return "CMS"; // 中小堆
        }
        return "G1"; // 默认推荐
    }
}
```

## 六、JVM参数配置

```java
public class GCParameters {

    // CMS参数
    // -XX:+UseConcMarkSweepGC     启用CMS
    // -XX:CMSInitiatingOccupancyFraction=70  触发阈值
    // -XX:+UseCMSInitiatingOccupancyOnly     使用固定阈值

    // G1参数
    // -XX:+UseG1GC                启用G1
    // -XX:MaxGCPauseMillis=200    最大停顿时间目标
    // -XX:G1HeapRegionSize=1m     Region大小

    // ZGC参数
    // -XX:+UseZGC                 启用ZGC
    // -XX:ZCollectionInterval=5   GC间隔

    public static void printGCConfig() {
        for (GarbageCollectorMXBean gc :
                ManagementFactory.getGarbageCollectorMXBeans()) {
            System.out.println("GC Name: " + gc.getName());
            System.out.println("  Collection Count: " + gc.getCollectionCount());
            System.out.println("  Collection Time:  " + gc.getCollectionTime() + "ms");
            System.out.println("  Memory Pools:     " +
                    String.join(", ", gc.getMemoryPoolNames()));
        }
    }
}
```

## 七、总结

垃圾收集器的演进体现了JVM对性能优化的持续追求。选择合适的收集器需要根据应用特点和性能需求来决定。

**学习要点**：
1. 掌握CMS的四个阶段和并发标记原理
2. 理解G1的Region划分和回收策略
3. 了解ZGC的着色指针和读屏障技术
4. 掌握不同场景下收集器的选择策略
