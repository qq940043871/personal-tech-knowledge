## 一、journalctl 日志管理

### 基础用法

```bash
# 查看所有日志
journalctl                               # 所有日志
journalctl -r                            # 倒序(最新在前)
journalctl -f                            # 实时追踪
journalctl -n 50                         # 最近50条

# 按时间过滤
journalctl --since "2026-05-19 10:00"    # 指定时间之后
journalctl --until "2026-05-19 11:00"    # 指定时间之前
journalctl --since "1 hour ago"          # 最近1小时
journalctl --since today                 # 今天
journalctl --since yesterday --until today  # 昨天

# 按服务过滤
journalctl -u nginx.service              # 指定服务
journalctl -u nginx.service -u php-fpm.service  # 多个服务
```

### 高级过滤

```bash
# 按优先级过滤
journalctl -p err                        # 错误级别
journalctl -p warning                    # 警告及以上
# 优先级: emerg(0) alert(1) crit(2) err(3) warning(4) notice(5) info(6) debug(7)

# 按进程过滤
journalctl _PID=1234                     # 指定PID
journalctl _UID=1000                     # 指定用户
journalctl _COMM=sshd                    # 指定命令名

# 按内核日志
journalctl -k                            # 内核日志
journalctl -k -b                         # 本次启动的内核日志
journalctl -b                            # 本次启动的所有日志
journalctl -b -1                         # 上次启动的日志

# 输出格式
journalctl -o json-pretty               # JSON格式
journalctl -o verbose                    # 详细格式
journalctl -o cat                        # 只输出消息
```

### 日志管理

```bash
# 查看日志占用空间
journalctl --disk-usage

# 清理日志
journalctl --vacuum-size=500M           # 保留500MB
journalctl --vacuum-time=30d            # 保留30天
journalctl --vacuum-files=10            # 保留10个文件

# 配置日志持久化
mkdir -p /var/log/journal
systemctl restart systemd-journald

# 配置文件 /etc/systemd/journald.conf
# [Journal]
# SystemMaxUse=500M
# SystemMaxFileSize=50M
# MaxRetentionSec=30day
```

## 二、syslog 日志

### 日志文件位置

```bash
# 常见日志文件
/var/log/syslog          # 系统日志(Debian/Ubuntu)
/var/log/messages        # 系统日志(CentOS/RHEL)
/var/log/auth.log        # 认证日志(Debian/Ubuntu)
/var/log/secure          # 认证日志(CentOS/RHEL)
/var/log/kern.log        # 内核日志
/var/log/cron.log        # 定时任务日志
/var/log/mail.log        # 邮件日志
/var/log/boot.log        # 启动日志

# 查看日志
tail -f /var/log/syslog                  # 实时追踪
cat /var/log/syslog | grep "error"       # 搜索错误
less /var/log/syslog                     # 分页查看
```

### rsyslog配置

```bash
# 配置文件 /etc/rsyslog.conf

# 日志规则格式
# facility.priority    action
# auth.*               /var/log/auth.log
# *.info               /var/log/syslog

# facility类型
# auth, authpriv, cron, daemon, kern, lpr, mail, news, syslog, user, local0-7

# 优先级
# emerg, alert, crit, err, warning, notice, info, debug

# 远程日志发送
# *.* @@remote-server:514               # TCP
# *.* @remote-server:514                # UDP

# 重启rsyslog
systemctl restart rsyslog
```

## 三、dmesg 内核日志

### 基础用法

```bash
# 查看内核日志
dmesg                                     # 所有内核日志
dmesg -T                                  # 显示时间戳
dmesg -T | tail -50                      # 最近50条
dmesg -T -l err                          # 只显示错误

# 实时监控
dmesg -T -w                              # 实时追踪
dmesg -T -w | grep -i "error\|warn"     # 追踪错误

# 按设施过滤
dmesg -f kern                            # 内核消息
dmesg -f user                            # 用户空间消息
```

### dmesg常见信息

```bash
# 硬件相关
dmesg | grep -i "usb"                    # USB设备
dmesg | grep -i "disk\|sda\|nvme"        # 磁盘
dmesg | grep -i "eth\|nic\|link"         # 网卡
dmesg | grep -i "memory\|ram"            # 内存
dmesg | grep -i "cpu\|processor"         # CPU

# OOM相关
dmesg | grep -i "oom\|out of memory\|killed"

# 错误信息
dmesg | grep -i "error\|fail\|warn" | tail -20

# 硬件错误
dmesg | grep -i "hardware error\|mce\|edac"
```

## 四、日志轮转

### logrotate配置

```bash
# 配置文件
/etc/logrotate.conf                       # 主配置
/etc/logrotate.d/                         # 子配置目录

# 查看logrotate状态
cat /var/lib/logrotate/status             # 轮转状态
logrotate -d /etc/logrotate.conf          # 调试模式(不实际执行)
```

### 配置示例

```bash
# /etc/logrotate.d/myapp
/var/log/myapp/*.log {
    daily                  # 每天轮转
    rotate 30              # 保留30个
    compress               # 压缩
    delaycompress          # 延迟压缩
    missingok              # 文件不存在不报错
    notifempty             # 空文件不轮转
    create 644 root root   # 创建新文件权限
    dateext                # 使用日期后缀
    dateformat -%Y%m%d     # 日期格式
    maxsize 100M           # 超过100M轮转
    sharedscripts          # 多文件共享脚本
    postrotate
        systemctl reload myapp
    endscript
}

# 手动执行轮转
logrotate -f /etc/logrotate.d/myapp      # 强制轮转
logrotate -v /etc/logrotate.d/myapp      # 详细输出
```

## 五、日志分析技巧

### 常用分析命令

```bash
# 统计错误数量
grep -c "ERROR" /var/log/syslog

# 按时间统计错误
grep "ERROR" /var/log/syslog | awk '{print $1, $2}' | uniq -c

# 提取IP地址
grep -oE '[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+' /var/log/auth.log | sort | uniq -c | sort -rn

# 统计登录失败
grep "Failed password" /var/log/auth.log | awk '{print $(NF-3)}' | sort | uniq -c | sort -rn

# 查找特定时间段日志
awk '/May 19 10:00/,/May 19 11:00/' /var/log/syslog
```

### 实时监控脚本

```bash
# 监控错误日志
tail -f /var/log/syslog | grep --line-buffered -i "error\|fail\|warn"

# 统计每分钟错误数
while true; do
    count=$(grep "ERROR" /var/log/syslog | grep "$(date '+%b %d %H:%M')" | wc -l)
    echo "$(date '+%H:%M:%S'): $count errors"
    sleep 60
done
```
