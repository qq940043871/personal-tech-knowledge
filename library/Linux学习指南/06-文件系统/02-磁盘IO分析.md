## 一、iostat 磁盘IO统计

### 基础用法

```bash
# 安装sysstat
apt install sysstat                       # Debian/Ubuntu
yum install sysstat                       # CentOS/RHEL

# 基础用法
iostat                                    # 一次性输出
iostat -x                                 # 扩展统计
iostat -x 1 5                             # 每秒刷新，共5次
iostat -x -d sda                          # 指定磁盘
iostat -x -p sda                          # 包含分区

# 关键指标
# Device: 设备名
# r/s: 每秒读请求数
# w/s: 每秒写请求数
# rkB/s: 每秒读取KB
# wkB/s: 每秒写入KB
# rrqm/s: 每秒读合并数
# wrqm/s: 每秒写合并数
# await: 平均IO等待时间(ms)
# svctm: 平均服务时间(ms)
# %util: 设备使用率
```

### iostat输出解读

```bash
# 关键阈值
# await > 10ms:   IO延迟较高
# await > 50ms:   IO延迟严重
# %util > 70%:    设备繁忙
# %util > 90%:    设备过载
# svctm > await:  可能有队列积压

# 实时监控
iostat -xm 1 | awk '/^sd/ && $NF > 50 {print $1, "util:", $NF"%", "await:", $(NF-2)"ms"}'

# 输出到文件
iostat -x 1 60 > /tmp/iostat_$(date +%Y%m%d_%H%M%S).log
```

## 二、iotop 进程IO监控

### 基础用法

```bash
# 安装iotop
apt install iotop

# 基础用法
iotop                                     # 交互式界面
iotop -o                                  # 只显示有IO的进程
iotop -b                                  # 批处理模式
iotop -P                                  # 只显示进程(不显示线程)
iotop -u username                         # 指定用户
iotop -p PID                              # 指定进程

# 批处理输出
iotop -b -o -n 10 > /tmp/iotop.log       # 输出10次

# 关键字段
# TID: 线程ID
# PRIO: 优先级
# READ: 读取速度
# WRITE: 写入速度
# SWAPIN: Swap使用率
# IO: IO等待率
# COMMAND: 命令名
```

### iotop交互命令

```bash
# iotop快捷键
# o: 只显示有IO的进程
# p: 切换进程/线程显示
# a: 切换累计/实时显示
# r: 反向排序
# ←→: 切换排序列
# q: 退出
```

## 三、fio 磁盘性能测试

### 基础测试

```bash
# 安装fio
apt install fio

# 顺序读测试
fio --name=seqread --rw=read --bs=128k --size=1G \
    --numjobs=1 --runtime=30 --filename=/tmp/fio_test

# 顺序写测试
fio --name=seqwrite --rw=write --bs=128k --size=1G \
    --numjobs=1 --runtime=30 --filename=/tmp/fio_test

# 随机读测试
fio --name=randread --rw=randread --bs=4k --size=1G \
    --numjobs=4 --runtime=30 --filename=/tmp/fio_test

# 随机写测试
fio --name=randwrite --rw=randwrite --bs=4k --size=1G \
    --numjobs=4 --runtime=30 --filename=/tmp/fio_test
```

### 混合负载测试

```bash
# 随机读写混合
fio --name=randrw --rw=randrw --rwmixread=70 --bs=4k \
    --size=1G --numjobs=4 --runtime=60 --filename=/tmp/fio_test

# 关键参数
# --name: 测试名称
# --rw: 读写模式(read/write/randread/randwrite/randrw)
# --bs: 块大小
# --size: 测试文件大小
# --numjobs: 并发数
# --runtime: 运行时间
# --filename: 测试文件路径

# 输出解读
# IOPS: 每秒IO操作数
# BW: 带宽(KB/s或MB/s)
# lat (usec/msec): 延迟
# slat: 提交延迟
# clat: 完成延迟
```

### fio测试场景

```bash
# 场景1: 数据库IO测试(4K随机读写)
fio --name=db --rw=randrw --rwmixread=70 --bs=4k \
    --size=10G --numjobs=8 --iodepth=32 --runtime=120 \
    --filename=/data/fio_test --direct=1

# 场景2: 大文件顺序读写
fio --name=largefile --rw=rw --bs=1M --size=10G \
    --numjobs=1 --runtime=60 --filename=/data/fio_test

# 场景3: 日志写入测试
fio --name=logwrite --rw=write --bs=4k --size=1G \
    --numjobs=1 --fsync=1 --runtime=60 --filename=/data/fio_test
```

## 四、IO调度器

### 查看与切换调度器

```bash
# 查看当前调度器
cat /sys/block/sda/queue/scheduler        # 查看当前调度器
cat /sys/block/sda/queue/nr_requests      # 查看队列深度

# 可用调度器
# noop: 无调度(适合SSD)
# deadline: 截止时间调度(适合数据库)
# cfq: 完全公平队列(默认，适合通用)
# mq-deadline: 多队列deadline(新内核)
# bfq: 预算公平队列
# kyber: 双队列调度

# 临时切换调度器
echo deadline > /sys/block/sda/queue/scheduler

# 永久切换: 内核参数
# GRUB_CMDLINE_LINUX="elevator=deadline"
# 或 udev规则
```

### 调度器选择建议

```bash
# SSD推荐: noop/mq-deadline
echo noop > /sys/block/nvme0n1/queue/scheduler

# HDD数据库: deadline/mq-deadline
echo deadline > /sys/block/sda/queue/scheduler

# HDD通用: cfq
echo cfq > /sys/block/sda/queue/scheduler

# 查看IO统计
cat /proc/diskstats                       # 磁盘统计
iostat -x 1                               # 实时统计
```

## 五、IO问题排查

```bash
# 步骤1: 确认IO瓶颈
iostat -x 1 | grep -E "Device|sd|nvme"
# 关注: %util, await, avgqu-sz

# 步骤2: 找出IO消耗进程
iotop -o -b -n 5

# 步骤3: 分析IO模式
# 随机IO: 小块、高IOPS
# 顺序IO: 大块、高吞吐

# 步骤4: 检查IO等待
vmstat 1 | awk 'NR>2 && $16>10 {print "IO wait high:", $16"%"}'

# 步骤5: 使用blktrace分析
blktrace -d /dev/sda -o - | blkparse -i -
# 或
btrace /dev/sda
```

## 六、磁盘健康监控

```bash
# 查看SMART信息
smartctl -a /dev/sda                      # 完整SMART信息
smartctl -H /dev/sda                      # 健康状态
smartctl -t short /dev/sda               # 短测试
smartctl -t long /dev/sda                # 长测试

# 关键SMART指标
# Reallocated_Sector_Ct: 重映射扇区
# Current_Pending_Sector: 待处理扇区
# Offline_Uncorrectable: 不可纠正扇区
# Temperature_Celsius: 温度

# 监控脚本
while true; do
    temp=$(smartctl -A /dev/sda | grep Temperature | awk '{print $10}')
    echo "$(date): Disk temp: ${temp}C"
    sleep 300
done
```
