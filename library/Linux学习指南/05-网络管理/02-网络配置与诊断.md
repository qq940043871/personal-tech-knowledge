## 一、ip 命令

### 网卡管理

```bash
# 查看网卡
ip addr show                              # 所有网卡
ip addr show eth0                         # 指定网卡
ip -br addr show                          # 简洁显示
ip -4 addr show                           # 只显示IPv4
ip -6 addr show                           # 只显示IPv6

# 配置IP地址
ip addr add 192.168.1.100/24 dev eth0    # 添加IP
ip addr del 192.168.1.100/24 dev eth0    # 删除IP

# 网卡状态
ip link show                              # 所有链路
ip link show eth0                         # 指定链路
ip link set eth0 up                       # 启用网卡
ip link set eth0 down                     # 禁用网卡
ip link set eth0 mtu 9000                # 设置MTU
ip link set eth0 promisc on              # 开启混杂模式
```

### 路由管理

```bash
# 查看路由
ip route show                             # 所有路由
ip route show table all                   # 所有路由表
ip route get 8.8.8.8                     # 查询到目标的路由

# 添加路由
ip route add 10.0.0.0/8 via 192.168.1.1  # 添加静态路由
ip route add default via 192.168.1.1     # 设置默认网关

# 删除路由
ip route del 10.0.0.0/8

# 策略路由
ip rule show                              # 查看规则
ip rule add from 192.168.1.0/24 table 100  # 添加规则
ip route add default via 10.0.0.1 table 100  # 路由表
```

### 邻居表（ARP）

```bash
# 查看邻居表
ip neigh show                             # ARP表
ip neigh show dev eth0                   # 指定网卡

# 添加静态ARP
ip neigh add 192.168.1.200 lladdr 00:11:22:33:44:55 dev eth0

# 删除ARP
ip neigh del 192.168.1.200 dev eth0

# 清除过期ARP
ip neigh flush all
```

## 二、ss 命令

### 连接查看

```bash
# 基础用法
ss -t                                     # TCP连接
ss -u                                     # UDP连接
ss -l                                     # 监听端口
ss -a                                     # 所有连接
ss -n                                     # 不解析服务名
ss -p                                     # 显示进程信息

# 组合用法
ss -tlnp                                  # TCP监听+进程
ss -tnp                                   # TCP已建立+进程
ss -tulnp                                 # TCP/UDP监听+进程
ss -ant                                   # 所有TCP(数字)
```

### 过滤与统计

```bash
# 状态过滤
ss -t state established                   # 已建立连接
ss -t state time-wait                     # TIME_WAIT
ss -t state close-wait                    # CLOSE_WAIT
ss -t state listen                        # 监听状态

# 地址过滤
ss -tn dst 192.168.1.100                 # 目标地址
ss -tn src 192.168.1.100                 # 源地址
ss -tn dport = 80                        # 目标端口
ss -tn sport = :80                       # 源端口

# 组合过滤
ss -tn '( dport = :80 or dport = :443 )'
ss -tn dst 192.168.1.100 and dport = 80

# 统计
ss -s                                     # 连接统计摘要
ss -ant | awk '{print $1}' | sort | uniq -c | sort -rn  # 状态统计
```

## 三、netstat 命令

### 基础用法

```bash
# 查看连接
netstat -tlnp                             # TCP监听+进程
netstat -ulnp                             # UDP监听+进程
netstat -anp                              # 所有连接+进程
netstat -ant                              # 所有TCP(数字)

# 路由表
netstat -rn                               # 路由表

# 网络统计
netstat -s                                # 协议统计
netstat -i                                # 网卡统计
netstat -i -e                             # 详细网卡统计

# 统计连接数
netstat -ant | awk '/^tcp/ {print $6}' | sort | uniq -c | sort -rn
netstat -ant | grep ":80" | wc -l        # 80端口连接数
```

## 四、tcpdump 抓包

### 基础抓包

```bash
# 基础用法
tcpdump -i eth0                           # 指定网卡
tcpdump -i any                            # 所有网卡
tcpdump -c 100                            # 抓100个包
tcpdump -nn                               # 不解析地址和端口

# 保存与读取
tcpdump -w capture.pcap                   # 保存到文件
tcpdump -w - | gzip > capture.gz         # 压缩保存
tcpdump -r capture.pcap                   # 读取文件
tcpdump -r capture.pcap -nn              # 读取(不解析)
```

### 过滤表达式

```bash
# 主机过滤
tcpdump host 192.168.1.100               # 源或目标
tcpdump src host 192.168.1.100           # 源地址
tcpdump dst host 192.168.1.100           # 目标地址

# 端口过滤
tcpdump port 80                           # 源或目标端口
tcpdump src port 80                       # 源端口
tcpdump dst port 80                       # 目标端口
tcpdump portrange 80-90                   # 端口范围

# 协议过滤
tcpdump tcp                               # TCP
tcpdump udp                               # UDP
tcpdump icmp                              # ICMP
tcpdump arp                               # ARP

# 组合过滤
tcpdump 'host 192.168.1.100 and port 80'
tcpdump 'tcp and (port 80 or port 443)'
tcpdump 'not port 22'                     # 排除SSH
```

### 高级抓包

```bash
# 显示内容
tcpdump -A -nn port 80                   # ASCII显示
tcpdump -X -nn port 80                   # 十六进制+ASCII
tcpdump -s 0 -A port 80                  # 完整包内容

# 抓取特定标志位
tcpdump 'tcp[tcpflags] & tcp-syn != 0'  # SYN包
tcpdump 'tcp[tcpflags] & tcp-fin != 0'  # FIN包
tcpdump 'tcp[tcpflags] & tcp-rst != 0'  # RST包

# 包大小过滤
tcpdump 'greater 1000'                    # 大于1000字节
tcpdump 'less 100'                        # 小于100字节

# 输出控制
tcpdump -tttt                             # 显示时间戳
tcpdump -e                                # 显示链路层头
tcpdump -v/-vv/-vvv                       # 详细程度
```

## 五、nc (netcat) 网络工具

### 基础用法

```bash
# 端口测试
nc -zv localhost 80                       # 测试端口连通性
nc -zv localhost 80-100                   # 测试端口范围
nc -zvn 192.168.1.100 80                 # 不解析DNS

# 监听端口
nc -l 8080                                # 监听端口
nc -lk 8080                               # 持续监听
nc -lp 8080                               # 指定端口监听

# 数据传输
# 发送方
echo "hello" | nc localhost 8080
# 接收方
nc -l 8080
```

### 高级用法

```bash
# 文件传输
# 接收方
nc -l 9999 > received_file
# 发送方
nc localhost 9999 < send_file

# 端口转发
nc -l 8080 | nc remote_host 80

# HTTP请求
echo -e "GET / HTTP/1.0\r\nHost: example.com\r\n\r\n" | nc example.com 80

# 简单聊天
# 终端1
nc -l 8080
# 终端2
nc localhost 8080

# 扫描端口
nc -zv 192.168.1.1 20-100                # 扫描20-100端口
```
