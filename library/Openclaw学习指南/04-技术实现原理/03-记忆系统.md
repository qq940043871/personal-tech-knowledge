# 记忆系统

## 概述

Openclaw 的记忆系统是其智能化的核心，让 AI 能够记住用户偏好、历史对话和上下文信息。

## 存储后端

### 本地文件存储

```typescript
class FileMemoryStore implements MemoryStore {
  private basePath: string
  
  constructor(basePath: string) {
    this.basePath = basePath
  }
  
  async add(entry: MemoryEntry): Promise<void> {
    const path = this.getPath(entry.id)
    await fs.writeFile(path, JSON.stringify(entry))
  }
  
  async get(id: string): Promise<MemoryEntry | null> {
    const path = this.getPath(id)
    try {
      const content = await fs.readFile(path, 'utf-8')
      return JSON.parse(content)
    } catch {
      return null
    }
  }
  
  async search(query: string, options?: SearchOptions): Promise<MemoryEntry[]> {
    const entries = await this.getAll()
    return this.fuzzySearch(entries, query, options)
  }
}
```

```typescript
class SQLiteMemoryStore implements MemoryStore {
  private db: Database
  
  async initialize(): Promise<void> {
    await this.db.exec(`
      CREATE TABLE IF NOT EXISTS memories (
        id TEXT PRIMARY KEY,
        content TEXT NOT NULL,
        type TEXT NOT NULL,
        embedding BLOB,
        metadata JSON,
        importance REAL,
        access_count INTEGER,
        last_accessed INTEGER,
        created_at INTEGER
      );
      
      CREATE INDEX IF NOT EXISTS idx_type ON memories(type);
      CREATE INDEX IF NOT EXISTS idx_importance ON memories(importance);
    `)
  }
  
  async add(entry: MemoryEntry): Promise<void> {
    await this.db.run(`
      INSERT INTO memories VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
    `, [
      entry.id,
      entry.content,
      entry.type,
      entry.embedding ? JSON.stringify(entry.embedding) : null,
      JSON.stringify(entry.metadata),
      entry.importance,
      entry.accessCount,
      entry.lastAccessed,
      Date.now()
    ])
  }
}
```

### 向量数据库存储

```typescript
class VectorMemoryStore implements MemoryStore {
  private client: VectorDBClient
  private embedder: Embedder
  
  async add(entry: MemoryEntry): Promise<void> {
    const embedding = await this.embedder.embed(entry.content)
    
    await this.client.upsert({
      id: entry.id,
      vector: embedding,
      metadata: {
        content: entry.content,
        type: entry.type,
        ...entry.metadata
      }
    })
  }
  
  async search(query: string, options?: SearchOptions): Promise<MemoryEntry[]> {
    const queryEmbedding = await this.embedder.embed(query)
    
    const results = await this.client.query({
      vector: queryEmbedding,
      topK: options?.limit || 10,
      filter: options?.filter
    })
    
    return results.map(r => ({
      id: r.id,
      content: r.metadata.content,
      type: r.metadata.type,
      embedding: r.vector,
      metadata: r.metadata,
      importance: r.score
    }))
  }
}
```

## 记忆检索

### 检索策略

```typescript
class MemoryRetriever {
  async retrieve(query: string, context: RetrievalContext): Promise<MemoryEntry[]> {
    const strategies = [
      this.semanticSearch(query),
      this.keywordSearch(query),
      this.recencySearch(context.timeWindow),
      this.importanceSearch(context.minImportance)
    ]
    
    const results = await Promise.all(strategies)
    return this.mergeAndRank(results)
  }
  
  private mergeAndRank(results: MemoryEntry[][]): MemoryEntry[] {
    const merged = new Map<string, MemoryEntry>()
    
    for (const list of results) {
      for (const entry of list) {
        if (merged.has(entry.id)) {
          const existing = merged.get(entry.id)!
          existing.importance = Math.max(existing.importance, entry.importance)
        } else {
          merged.set(entry.id, entry)
        }
      }
    }
    
    return Array.from(merged.values())
      .sort((a, b) => b.importance - a.importance)
  }
}
```

### 上下文注入

```typescript
class MemoryInjector {
  async inject(context: ConversationContext): Promise<ConversationContext> {
    const relevantMemories = await this.retrieve(
      context.currentQuery,
      { maxEntries: 10 }
    )
    
    const memoryBlock = this.formatMemories(relevantMemories)
    
    return {
      ...context,
      systemPrompt: context.systemPrompt + '\n\n' + memoryBlock
    }
  }
  
  private formatMemories(memories: MemoryEntry[]): string {
    if (memories.length === 0) return ''
    
    return `## 相关记忆
${memories.map(m => `- [${m.type}] ${m.content}`).join('\n')}`
  }
}
```

## 记忆管理

### 记忆更新

```typescript
class MemoryManager {
  async update(id: string, updates: Partial<MemoryEntry>): Promise<void> {
    const entry = await this.store.get(id)
    if (!entry) throw new MemoryNotFoundError(id)
    
    const updated = { ...entry, ...updates }
    await this.store.add(updated)
  }
  
  async incrementAccess(id: string): Promise<void> {
    const entry = await this.store.get(id)
    if (!entry) return
    
    await this.update(id, {
      accessCount: entry.accessCount + 1,
      lastAccessed: Date.now()
    })
  }
}
```

### 记忆遗忘

```typescript
class MemoryForgetter {
  private threshold = 0.1
  private maxAge = 30 * 24 * 60 * 60 * 1000  // 30天
  
  async forget(): Promise<void> {
    const all = await this.store.getAll()
    
    for (const entry of all) {
      if (this.shouldForget(entry)) {
        await this.store.delete(entry.id)
      }
    }
  }
  
  private shouldForget(entry: MemoryEntry): boolean {
    const age = Date.now() - entry.metadata.timestamp
    const isOld = age > this.maxAge
    const isUnimportant = entry.importance < this.threshold
    const isUnused = entry.accessCount === 0
    
    return isOld && isUnimportant && isUnused
  }
}
```

```typescript
class SoulLoader {
  async load(): Promise<string> {

      const content = await fs.readFile(soulPath, 'utf-8')
```

## 隐私与安全

### 敏感信息过滤

```typescript
class SensitiveFilter {
  private patterns = [
    /password\s*[:=]\s*\S+/gi,
    /api[_-]?key\s*[:=]\s*\S+/gi,
    /secret\s*[:=]\s*\S+/gi,
    /\b\d{16,19}\b/g,  // 信用卡号
    /\b[\w.-]+@[\w.-]+\.\w+\b/g  // 邮箱
  ]
  
  filter(content: string): string {
    let filtered = content
    for (const pattern of this.patterns) {
      filtered = filtered.replace(pattern, '[REDACTED]')
    }
    return filtered
  }
}
```

### 记忆加密

```typescript
class EncryptedMemoryStore implements MemoryStore {
  private store: MemoryStore
  private cipher: Cipher
  
  async add(entry: MemoryEntry): Promise<void> {
    const encrypted = {
      ...entry,
      content: this.cipher.encrypt(entry.content)
    }
    await this.store.add(encrypted)
  }
  
  async get(id: string): Promise<MemoryEntry | null> {
    const entry = await this.store.get(id)
    if (!entry) return null
    
    return {
      ...entry,
      content: this.cipher.decrypt(entry.content)
    }
  }
}
```
