  return arr.filter(item => {

  })

### 错误分析

```
用户: 分析这个错误：TypeError: Cannot read property 'map' of undefined

Openclaw: 错误分析：

**错误类型：** TypeError
**原因：** 尝试对 undefined 调用 .map() 方法

**可能原因：**
1. API 返回数据为空或格式错误
2. 异步数据未加载完成就渲染
3. 变量名拼写错误

**修复建议：**

```typescript
// 问题代码
function UserList({ users }) {
  return users.map(user => <div>{user.name}</div>)
}

// 修复方案 1：添加默认值
function UserList({ users = [] }) {
  return users.map(user => <div>{user.name}</div>)
}

// 修复方案 2：条件渲染
function UserList({ users }) {
  if (!users) return <div>Loading...</div>
  return users.map(user => <div>{user.name}</div>)
}

// 修复方案 3：可选链
function UserList({ users }) {
  return users?.map(user => <div>{user.name}</div>) ?? null
}
```
