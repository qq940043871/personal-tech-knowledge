//日志格式 中间件|开始时间|协议号|耗时|结果|原因|
-- ._|-- ::|K|||

//统计当日总协议数
cat mointor.log|wc -l

//分协议号统计调用次数、并排序
cat monintor.log|gawk -F'|' '{print $ $}'|gawk '{sum[$]+=}END{for(i in sum)print i "\t" sum[i]}'|sort -nr -k

//统计每秒钟协议调用次数
cat monintor.log|gawk -F'|' '{print $ $ $}'|gawk '{sum[$,$]+=}END{for(i in sum)print i "\t" sum[i]}'|awk '$=="PA"{print $}'|sort -n -k
