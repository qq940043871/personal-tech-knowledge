## 权限模型

### 权限类型

| 权限 | 说明 | 风险级别 |
|------|------|----------|
| `fs:read` | 文件读取 | 低 |
| `fs:write` | 文件写入 | 中 |
| `fs:delete` | 文件删除 | 高 |
| `shell:execute` | 命令执行 | 高 |
| `network:http` | HTTP 请求 | 中 |
| `network:ws` | WebSocket 连接 | 中 |
| `process:spawn` | 进程创建 | 高 |
| `memory:read` | 记忆读取 | 低 |
| `memory:write` | 记忆写入 | 中 |

## 沙箱隔离

### 文件系统沙箱

```typescript
class SandboxFileSystem {
  private allowedPaths: string[]
  private deniedPaths: string[]
  
  async readFile(path: string): Promise<string> {
    const resolved = this.resolvePath(path)
    
    if (!this.isAllowed(resolved)) {
      throw new PermissionDeniedError(`Cannot read: ${path}`)
    }
    
    return fs.readFile(resolved, 'utf-8')
  }
  
  async writeFile(path: string, content: string): Promise<void> {
    const resolved = this.resolvePath(path)
    
    if (!this.isAllowed(resolved, 'write')) {
      throw new PermissionDeniedError(`Cannot write: ${path}`)
    }
    
    return fs.writeFile(resolved, content)
  }
  
  private isAllowed(path: string, mode: 'read' | 'write' = 'read'): boolean {
    // 检查黑名单
    for (const denied of this.deniedPaths) {
      if (minimatch(path, denied)) return false
    }
    
    // 检查白名单
    for (const allowed of this.allowedPaths) {
      if (minimatch(path, allowed)) return true
    }
    
    return false
  }
}
```

### Shell 沙箱

```typescript
class SandboxShell {
  private allowedCommands: string[]
  private deniedPatterns: RegExp[]
  
  async execute(command: string): Promise<ShellResult> {
    const parsed = this.parseCommand(command)
    
    // 检查命令是否在允许列表
    if (!this.allowedCommands.includes(parsed.name)) {
      throw new PermissionDeniedError(`Command not allowed: ${parsed.name}`)
    }
    
    // 检查危险模式
    for (const pattern of this.deniedPatterns) {
      if (pattern.test(command)) {
        throw new SecurityError(`Dangerous pattern detected: ${pattern}`)
      }
    }
    
    return this.runInSandbox(parsed)
  }
  
  private async runInSandbox(cmd: ParsedCommand): Promise<ShellResult> {
    // 使用子进程隔离执行
    const child = spawn(cmd.name, cmd.args, {
      cwd: this.sandboxDir,
      env: this.filteredEnv,
      timeout: this.timeout,
      maxBuffer: this.maxBuffer
    })
    
    return this.collectOutput(child)
  }
}
```

### 网络沙箱

```typescript
class SandboxNetwork {
  private allowedDomains: string[]
  private deniedIPs: string[]
  
  async fetch(url: string, options?: FetchOptions): Promise<Response> {
    const parsed = new URL(url)
    
    // 检查域名白名单
    if (!this.isDomainAllowed(parsed.hostname)) {
      throw new PermissionDeniedError(`Domain not allowed: ${parsed.hostname}`)
    }
    
    // 检查是否为内网 IP
    if (this.isPrivateIP(parsed.hostname)) {
      throw new SecurityError('Cannot access private network')
    }
    
    return fetch(url, options)
  }
  
  private isPrivateIP(hostname: string): boolean {
    const ip = dns.lookup(hostname)
    return ip.startsWith('192.168.') ||
           ip.startsWith('10.') ||
           ip.startsWith('172.16.')
  }
}
```

### 2. 敏感数据保护

```typescript
class SensitiveDataProtector {
  private patterns = [
    { name: 'api_key', pattern: /(?:api[_-]?key|apikey)[\s:=]+['"]?([a-zA-Z0-9_-]+)/gi },
    { name: 'password', pattern: /password[\s:=]+['"]?([^\s'"]+)/gi },
    { name: 'token', pattern: /token[\s:=]+['"]?([a-zA-Z0-9_-]+)/gi }
  ]
  
  scan(content: string): SensitiveData[] {
    const findings: SensitiveData[] = []
    
    for (const { name, pattern } of this.patterns) {
      let match
      while ((match = pattern.exec(content)) !== null) {
        findings.push({
          type: name,
          value: match[1],
          position: match.index
        })
      }
    }
    
    return findings
  }
  
  redact(content: string): string {
    let redacted = content
    for (const { pattern } of this.patterns) {
      redacted = redacted.replace(pattern, '$1[REDACTED]')
    }
    return redacted
  }
}
```

### 4. 安全事件响应

```typescript
class SecurityEventHandler {
  private handlers: Map<SecurityEvent, EventHandler>
  
  constructor() {
    this.handlers
      .set('permission_denied', this.handlePermissionDenied)
      .set('suspicious_activity', this.handleSuspiciousActivity)
      .set('rate_limit_exceeded', this.handleRateLimitExceeded)
  }
  
  async handle(event: SecurityEvent): Promise<void> {
    const handler = this.handlers.get(event.type)
    if (handler) {
      await handler(event)
    }
    
    await this.alert(event)
    await this.log(event)
  }
  
  private async alert(event: SecurityEvent): Promise<void> {
    if (event.severity >= Severity.HIGH) {
      await this.notifyAdmins(event)
    }
  }
}
```
